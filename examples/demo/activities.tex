% texstage's own kinds of live activities: a choice, a number and a short answer.
% They run on the activity framework's core: texstage present examples/demo/activities.tex --activities core
\documentclass[aspectratio=169]{beamer}
\usetheme{default}
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{footline}[frame number]
\usepackage{texquiz}       % the join frame
\usepackage{texactivity}
\quizsetup{join-url={https://quiz.example.org/play}}

\begin{document}

\begin{frame}{Knowledge Graphs, live}
  A choice, a number and a short answer: three live activities on your phones.
  \note{Welcome everyone. This deck needs a server started with --activities core.}
\end{frame}

\quizjoinframe[note={Phones out: scan the code, enter the PIN and a nickname.}]

\begin{frame}[fragile]{Writing RDF down}
  \begin{activity}[id=rdf-syntaxes, kind=choice, select=many,
      options={Turtle, JSON-LD, SPARQL, N-Triples}, height=.6\textheight]
    \prompt{Which of them write down RDF graphs? Choose all that do.}
    \reference{Turtle}
    \reference{JSON-LD}
    \reference{N-Triples}
  \end{activity}
  \note{SPARQL queries graphs; it does not write them down. Reveal from the Activities panel.}
\end{frame}

\begin{frame}[fragile]{Count the triples}
  \texttt{:alice :knows :bob, :carol ; :name "Alice" .}
  \medskip
  \begin{activity}[id=turtle-count, kind=number, height=.5\textheight]
    \prompt{How many triples does this Turtle statement write?}
    \reference{3}
  \end{activity}
  \note{The comma repeats subject and predicate, the semicolon the subject: three triples.}
\end{frame}

\begin{frame}[fragile]{Asking a graph}
  \begin{activity}[id=query-language, kind=text, height=.6\textheight]
    \prompt{One word: which W3C language queries RDF graphs?}
    \reference{SPARQL}
  \end{activity}
  \note{Case and spacing do not matter: sparql counts too.}
\end{frame}

\begin{frame}{Summary}
  An ordinary slide after the activities.
\end{frame}

\end{document}
