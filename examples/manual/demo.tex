\begin{quiz}[id=planet]{Which planet is known as the Red Planet?}
  \choice{Venus}
  \choice*{Mars}
  \choice{Jupiter}
  \choice{Neptune}
  \explain{Mars appears reddish because of iron oxide on its surface.}
  \note{On the reading slide, let the group read first. Advance once more to display the choices and start the timed question. Discuss the explanation after revealing the answer.}
\end{quiz}

\begin{quiz}[id=tools]{Which tools help you present a lesson?}
  \choice*{Speaker notes}
  \choice{Sharing the presenter key with the audience}
  \choice*{A separate projector window}
  \choice{Showing the correct answers before the question}
  \explain{Speaker notes and a separate projector window support the presenter while keeping private controls out of the audience view.}
\end{quiz}
